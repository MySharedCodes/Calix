cmake_minimum_required(VERSION 3.24)
project(calix_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  add_executable(calix-cli cli_launcher.cpp)
  if(MSVC)
    target_compile_options(calix-cli PRIVATE /O2 /EHsc /W4 /utf-8)
    set_property(TARGET calix-cli PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
if(NOT DEFINED BOOST_ROOT)
  message(FATAL_ERROR "Set BOOST_ROOT to the extracted Boost 1.86.0 source directory")
endif()
add_library(calix_avx2 OBJECT simd.cpp)
target_compile_definitions(calix_avx2 PRIVATE CALIX_SIMD_WIDTH=4)
add_library(calix_avx512 OBJECT simd.cpp)
target_compile_definitions(calix_avx512 PRIVATE CALIX_SIMD_WIDTH=8)
add_library(calix_amx OBJECT amx.cpp)
add_library(calix_native SHARED kernel.cpp fem.cpp triangle.cpp lattice.cpp cpu.cpp parallel.cpp sparse_runtime.cpp $<TARGET_OBJECTS:calix_avx2> $<TARGET_OBJECTS:calix_avx512> $<TARGET_OBJECTS:calix_amx>)
target_include_directories(calix_native SYSTEM PRIVATE "${BOOST_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/eigen-3.4.0")
target_compile_definitions(calix_native PRIVATE EIGEN_MPL2_ONLY)
target_compile_definitions(calix_native PRIVATE BOOST_ALL_NO_LIB NOMINMAX)
if(MSVC)
  target_compile_options(calix_avx2 PRIVATE /O2 /EHsc /W4 /arch:AVX2 /fp:strict)
  target_compile_options(calix_avx512 PRIVATE /O2 /EHsc /W4 /arch:AVX512 /fp:strict)
  target_compile_options(calix_amx PRIVATE /O2 /EHsc /W4 /fp:strict)
  foreach(target calix_avx2 calix_avx512 calix_amx)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endforeach()
  target_compile_options(calix_native PRIVATE /O2 /EHsc /W4 /utf-8)
  set_property(TARGET calix_native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(calix_native PRIVATE -O3 -Wall -Wextra)
endif()
